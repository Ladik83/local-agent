function makeFish(col){
  const g=new THREE.Group();
  const bodyMat=new THREE.MeshStandardMaterial({color:col,roughness:0.4});
  const body=new THREE.Mesh(new THREE.SphereGeometry(0.22,12,12),bodyMat);
  body.scale.set(1,1.4,0.7); g.add(body);
  const tail=new THREE.Mesh(new THREE.ConeGeometry(0.12,0.35,8),bodyMat);
  tail.position.z=-0.35; tail.rotation.x=Math.PI/2; g.add(tail);
  const fin=new THREE.Mesh(new THREE.ConeGeometry(0.08,0.2,6),bodyMat);
  fin.position.set(0,0.18,0); fin.rotation.x=-Math.PI/2; g.add(fin);
  const eye=new THREE.Mesh(new THREE.SphereGeometry(0.04,8,8),new THREE.MeshStandardMaterial({color:0x111111}));
  eye.position.set(0.18,0.05,0.15); g.add(eye);
  g.userData={tail:tail,speed:0.6+Math.random()*0.6};
  return g;
}

function updateCamera(){
  const sinp=Math.sin(phi), cosp=Math.cos(phi);
  camera.position.set(radius*Math.sin(phi)*Math.sin(theta), radius*cosp, radius*Math.sin(phi)*Math.cos(theta));
  camera.lookAt(0,0,0);
}

let bubbleIndex=0;
function spawnBubble(){
  const b=bubbles[bubbleIndex% bubbles.length]; bubbleIndex++;
  b.position.copy(pump.position); b.position.y+=1.0; b.position.x+=(Math.random()-0.5)*0.1; b.position.z+=(Math.random()-0.5)*0.1;
  b.visible=true; b.scale.setScalar(0.5+Math.random()*0.7);
}

function animate(){
  requestAnimationFrame(animate);
  if(!paused){
    t+=0.016;
    causticMat.uniforms.time.value=t*0.3;
    // вода рябь
    water.material.opacity=0.35+Math.sin(t*1.5)*0.03;
    // растения
    plants.forEach(p=>{p.rotation.z=Math.sin(t*p.userData.freq+p.userData.phase)*p.userData.amp;});
    // пузырьки
    if(Math.floor(t*2)%1===0) spawnBubble();
    bubbles.forEach(b=>{if(!b.visible)return; b.position.y+=b.userData.vy; b.position.x+=Math.sin(t+b.position.y)*0.002; if(b.position.y>water.position.y){b.visible=false;}});
    // рыбы
    fishes.forEach(f=>{
      const u=f.userData;
      const x=u.base.x+Math.sin(t*u.fx+u.px)*u.ampX+Math.sin(t*u.fx2+u.px)*u.ampX2;
      const y=u.base.y+Math.sin(t*u.fy+u.py)*u.ampY;
      const z=u.base.z+Math.sin(t*u.fz+u.pz)*u.ampZ;
      f.position.set(x,y,z);
      const nx=x+Math.sin((t+0.1)*u.fx+u.px)*u.ampX, ny=y+Math.sin((t+0.1)*u.fy+u.py)*u.ampY, nz=z+Math.sin((t+0.1)*u.fz+u.pz)*u.ampZ;
      f.lookAt(nx,ny,nz);
      f.children[1].rotation.y=Math.sin(t*u.speed*6)*0.6;
    });
    // краб
    crab.position.x=Math.sin(t*0.3)*2.2;
    crab.position.z=Math.cos(t*0.22)*1.8;
    crab.rotation.y=Math.sin(t*0.5)*0.4;
    // улитки
    snails.forEach(s=>{
      s.position.x+=s.userData.dir.x*s.userData.speed*0.01;
      s.position.z+=s.userData.dir.y*s.userData.speed*0.01;
      if(Math.abs(s.position.x)>W/2-0.5||Math.abs(s.position.z)>D/2-0.5){s.userData.dir.set(Math.random()-0.5,Math.random()-0.5).normalize();}
      s.rotation.y=Math.atan2(s.userData.dir.y,s.userData.dir.x);
    });
  }
  renderer.render(scene,camera);
}
